   Before:
   -------
   class Controller {
   public:
       void iniciar(...);
       void crearProceso(...);
       ...
   };

   After:
   ------
   class Controller : public IMotorSimulacion {
   public:
       void iniciar(...) override;
       void crearProceso(...) override;
       float obtenerUsoCPU() const override;
       ...
   };

   InterfazGrafica gui(controller_ptr);  // controller_ptr es IMotorSimulacion*

📋 PRÓXIMOS PASOS (A EJECUTAR)
═══════════════════════════════════════════════════════════════════════════════

1. ✅ Actualizar Controller para hereda de IMotorSimulacion

   Editar src/controller/Controller.h:
   ────────────────────────────────────
   #pragma once
   #include "../view/IMotorSimulacion.h"

   namespace sim {

   class Controller : public IMotorSimulacion {
   public:
       explicit Controller(Model& model);

       void iniciar(const std::string& algoritmo, int quantum) override;
       void crearProceso(const std::string& nombre, uint64_t rafaga, int prioridad, uint32_t memoria) override;
       void ejecutarPasoSiguiente() override;
       void forzarSalidaProceso(uint32_t pid, const std::string& causa) override;
       void cambiarEstadoSuspension(uint32_t pid, bool suspender) override;

       float obtenerUsoCPU() const override { return 25.0f; }  // Demo
       float obtenerUsoMemoria() const override { return 512.0f; }  // Demo
       uint32_t obtenerTotalProcesos() const override;

   private:
       Model& model_;
   };

   } // namespace sim

2. Crear archivo main_wx.cpp

   Estructura:
   ───────────
   #include <wx/wx.h>
   #include "src/view/InterfazGrafica.h"
   #include "src/controller/Controller.h"
   #include "src/model/Model.h"

   class SimuladorApp : public wxApp {
   public:
       bool OnInit() override;
   };

   bool SimuladorApp::OnInit() {
       sim::Model model;
       sim::Controller controller(model);

       sim::InterfazGrafica* gui = new sim::InterfazGrafica(&controller);
       gui->Show();

       return true;
   }

   wxIMPLEMENT_APP(SimuladorApp);

3. Actualizar CMakeLists.txt para wxWidgets

   Añadir:
   ───────
   find_package(wxWidgets REQUIRED COMPONENTS core base)
   include(${wxWidgets_USE_FILE})

   add_executable(simulador_wx
       src/main_wx.cpp
       src/model/Model.cpp
       src/controller/Controller.cpp
       src/view/InterfazGrafica.cpp
   )

   target_link_libraries(simulador_wx PRIVATE ${wxWidgets_LIBRARIES})

4. Compilar y ejecutar

   Comandos:
   ─────────
   cmake -G "MinGW Makefiles" -S . -B build
   cmake --build build --config Release
   .\build\simulador_wx.exe
